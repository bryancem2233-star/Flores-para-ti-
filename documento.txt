  <script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');

    let width = canvas.width = window.innerWidth;
    let height = canvas.height = window.innerHeight;

    window.addEventListener('resize', () => {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    });

    // Arreglo de flores
    const flores = [];
    let animacionActiva = false;

    class Flor {
      constructor(x, y, esCayendo = false) {
        this.x = x || Math.random() * width;
        this.y = y || (esCayendo ? -50 : height + 50);
        this.tamano = Math.random() * 25 + 15; // Tamaño del centro
        this.numPetalos = Math.floor(Math.random() * 4) + 12; // 12 a 15 pétalos
        this.velocidadY = esCayendo ? Math.random() * 2 + 1 : -(Math.random() * 2 + 1);
        this.velocidadX = Math.sin(Math.random() * Math.PI) * 1.5;
        this.rotacion = Math.random() * Math.PI * 2;
        this.velRotacion = (Math.random() - 0.5) * 0.03;
        this.colorCentro = '#5c3a21';
        this.colorPetalos = Math.random() > 0.3 ? '#ffd700' : '#ffae00'; // Amarillo / Dorado
        this.esCayendo = esCayendo;
        this.opacidad = Math.random() * 0.5 + 0.5;
      }

      dibujar() {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate(this.rotacion);
        ctx.globalAlpha = this.opacidad;

        // Dibujar Pétalos
        const largoPetalo = this.tamano * 1.8;
        const anchoPetalo = this.tamano * 0.6;

        for (let i = 0; i < this.numPetalos; i++) {
          const angulo = (i * 2 * Math.PI) / this.numPetalos;
          ctx.save();
          ctx.rotate(angulo);

          ctx.beginPath();
          ctx.fillStyle = this.colorPetalos;
          ctx.moveTo(0, 0);
          ctx.quadraticCurveTo(anchoPetalo, largoPetalo / 2, 0, largoPetalo);
          ctx.quadraticCurveTo(-anchoPetalo, largoPetalo / 2, 0, 0);
          ctx.fill();

          // Sombra suave en el pétalo
          ctx.strokeStyle = '#e69500';
          ctx.lineWidth = 1;
          ctx.stroke();

          ctx.restore();
        }

        // Dibujar Centro de la Flor (Girasol)
        ctx.beginPath();
        ctx.arc(0, 0, this.tamano, 0, Math.PI * 2);
        ctx.fillStyle = this.colorCentro;
        ctx.fill();
        ctx.strokeStyle = '#3d2300';
        ctx.lineWidth = 2;
        ctx.stroke();

        // Detalle del centro
        ctx.beginPath();
        ctx.arc(0, 0, this.tamano * 0.7, 0, Math.PI * 2);
        ctx.fillStyle = '#3d2300';
        ctx.fill();

        ctx.restore();
      }

      actualizar() {
        this.y += this.velocidadY;
        this.x += Math.sin(this.y * 0.01) + this.velocidadX;
        this.rotacion += this.velRotacion;

        // Si sale de pantalla, reiniciarla
        if (this.esCayendo && this.y > height + 100) {
          this.y = -50;
          this.x = Math.random() * width;
        } else if (!this.esCayendo && this.y < -100) {
          this.y = height + 50;
          this.x = Math.random() * width;
        }
      }
    }

    // Función para crear lluvia masiva de flores
    function generarFloresMasivas() {
      // 80 flores cayendo desde arriba y subiendo desde abajo
      for (let i = 0; i < 40; i++) {
        flores.push(new Flor(Math.random() * width, Math.random() * height, true));
        flores.push(new Flor(Math.random() * width, Math.random() * height, false));
      }
    }

    // Bucle de animación continuo
    function animar() {
      ctx.clearRect(0, 0, width, height);

      // Fondo dinámico estrellado / destellos
      for (let i = 0; i < flores.length; i++) {
        flores[i].actualizar();
        flores[i].dibujar();
      }

      requestAnimationFrame(animar);
    }

    // Reproducir melodía suave con la Web Audio API
    function tocarMelodia() {
      const AudioContext = window.AudioContext || window.webkitAudioContext;
      if (!AudioContext) return;
      const audioCtx = new AudioContext();

      const notas = [261.63, 329.63, 392.00, 523.25, 440.00, 349.23, 329.63, 392.00];
      let i = 0;

      setInterval(() => {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();

        osc.type = 'triangle';
        osc.frequency.setValueAtTime(notas[i % notas.length], audioCtx.currentTime);

        gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 1.2);

        osc.connect(gain);
        gain.connect(audioCtx.destination);

        osc.start();
        osc.stop(audioCtx.currentTime + 1.2);
        i++;
      }, 500);
    }

    // Iniciar experiencia al presionar el botón
    function iniciarPrograma() {
      document.getElementById('startBox').classList.add('hidden');
      document.getElementById('banner').classList.add('show');

      generarFloresMasivas();
      if (!animacionActiva) {
        animacionActiva = true;
        animar();
      }
      tocarMelodia();
    }
  </script>
